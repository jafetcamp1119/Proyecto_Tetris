CREATE VIEW VistaRanking
AS

SELECT
    U.UsuarioId,
    U.NombreUsuario,
    E.PartidasJugadas,
    E.PartidasGanadas,
    E.PartidasPerdidas,
    E.PuntuacionMaxima,
    E.TotalPuntos,
    E.LineasEliminadas,
    E.NivelMaximoAlcanzado

FROM Usuarios U

INNER JOIN EstadisticasJugador E
    ON U.UsuarioId = E.UsuarioId

WHERE U.Activo = 1;

CREATE OR ALTER PROCEDURE dbo.DeterminarGanador
    @PartidaId int
AS
BEGIN
    SET NOCOUNT ON;
    SELECT r.UsuarioId AS GanadorId
    FROM dbo.Resultados r
    INNER JOIN dbo.Partidas p ON p.PartidaId = r.PartidaId
    WHERE r.PartidaId = @PartidaId AND p.Estado = N'Finalizada' AND r.EsGanador = 1
    ORDER BY r.UsuarioId;
END;
GO

CREATE PROCEDURE EliminarColumna
    @PartidaId INT,
    @UsuarioAtacanteId INT,
    @Columna INT
AS
BEGIN

    SET NOCOUNT ON;

    DECLARE @TableroId INT;
    DECLARE @CantidadEliminada INT;

    SELECT
        @TableroId = TableroId

    FROM TablerosPartida

    WHERE PartidaId = @PartidaId;

    SELECT
        @CantidadEliminada = COUNT(*)

    FROM BloquesTablero

    WHERE TableroId = @TableroId
      AND Columna = @Columna
      AND UsuarioId <> @UsuarioAtacanteId
      AND Activo = 1;

    UPDATE BloquesTablero

    SET Activo = 0

    WHERE TableroId = @TableroId
      AND Columna = @Columna
      AND UsuarioId <> @UsuarioAtacanteId
      AND Activo = 1;

    INSERT INTO EliminacionesBloques
    (
        PartidaId,
        UsuarioAtacanteId,
        ColumnaAfectada,
        CantidadBloquesEliminados,
        PuntosObtenidos
    )
    VALUES
    (
        @PartidaId,
        @UsuarioAtacanteId,
        @Columna,
        @CantidadEliminada,
        @CantidadEliminada * 10
    );

    UPDATE ParticipantesPartida

    SET Puntuacion =
        Puntuacion + (@CantidadEliminada * 10)

    WHERE PartidaId = @PartidaId
      AND UsuarioId = @UsuarioAtacanteId;

    SELECT
        @CantidadEliminada AS BloquesEliminados;

END;
GO
